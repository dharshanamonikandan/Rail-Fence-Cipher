<script>
  const RAIL_COLORS = ['#7F77DD','#1D9E75','#D85A30','#378ADD','#BA7517','#D4537E','#639922'];
  const RAIL_BG    = ['#26215C','#085041','#4A1B0C','#042C53','#412402','#4B1528','#173404'];

  function switchTab(t) {
    document.getElementById('tab-enc').classList.toggle('active', t === 'enc');
    document.getElementById('tab-dec').classList.toggle('active', t === 'dec');
    document.getElementById('panel-enc').style.display = t === 'enc' ? '' : 'none';
    document.getElementById('panel-dec').style.display = t === 'dec' ? '' : 'none';
  }

  function clearResult(mode) {
    document.getElementById(mode + '-result').style.display = 'none';
    document.getElementById(mode + '-error').textContent = '';
  }

  // ── Encryption ──────────────────────────────────
  function encrypt(text, rails) {
    const n = text.length;
    const grid = Array.from({ length: rails }, () => Array(n).fill(''));
    let row = 0, dir = 1;
    for (let i = 0; i < n; i++) {
      grid[row][i] = text[i];
      if (row === 0) dir = 1;
      else if (row === rails - 1) dir = -1;
      row += dir;
    }
    return grid.map(r => r.filter(c => c !== '').join('')).join('');
  }

  // ── Decryption ──────────────────────────────────
  function decrypt(cipher, rails) {
    const n = cipher.length;
    const grid = Array.from({ length: rails }, () => Array(n).fill(''));
    let row = 0, dir = 1;
    for (let i = 0; i < n; i++) {
      grid[row][i] = '*';
      if (row === 0) dir = 1;
      else if (row === rails - 1) dir = -1;
      row += dir;
    }
    let idx = 0;
    for (let r = 0; r < rails; r++)
      for (let c = 0; c < n; c++)
        if (grid[r][c] === '*') grid[r][c] = cipher[idx++];
    const plain = [];
    row = 0; dir = 1;
    for (let i = 0; i < n; i++) {
      plain.push(grid[row][i]);
      if (row === 0) dir = 1;
      else if (row === rails - 1) dir = -1;
      row += dir;
    }
    return plain.join('');
  }

  // ── Zigzag visualizer ───────────────────────────
  function buildViz(text, rails) {
    const n = text.length;
    const positions = [];
    let row = 0, dir = 1;
    for (let i = 0; i < n; i++) {
      positions.push(row);
      if (row === 0) dir = 1;
      else if (row === rails - 1) dir = -1;
      row += dir;
    }
    const maxShow = 50;
    const showN = Math.min(n, maxShow);
    let html = '<table class="viz-table"><tbody>';
    for (let r = 0; r < rails; r++) {
      const fg = RAIL_COLORS[r % RAIL_COLORS.length];
      const bg = RAIL_BG[r % RAIL_BG.length];
      html += '<tr>';
      for (let i = 0; i < showN; i++) {
        if (positions[i] === r) {
          html += `<td><span class="viz-char" style="background:${bg};color:${fg}">${text[i]}</span></td>`;
        } else {
          html += `<td><span class="viz-dot">·</span></td>`;
        }
      }
      if (n > maxShow) {
        html += `<td style="font-size:11px;color:#555;padding-left:6px">+${n - maxShow}</td>`;
      }
      html += '</tr>';
    }
    html += '</tbody></table>';
    return html;
  }

  function buildLegend(rails) {
    let html = '';
    for (let r = 0; r < rails; r++) {
      const fg = RAIL_COLORS[r % RAIL_COLORS.length];
      const bg = RAIL_BG[r % RAIL_BG.length];
      html += `<span class="rail-badge" style="background:${bg};color:${fg}">rail ${r + 1}</span>`;
    }
    return html;
  }

  // ── Run encrypt ─────────────────────────────────
  function runEncrypt() {
    const text  = document.getElementById('enc-input').value.trim();
    const rails = parseInt(document.getElementById('enc-rails').value);
    const errEl = document.getElementById('enc-error');
    errEl.textContent = '';
    if (!text)                        { errEl.textContent = '[!] Plaintext cannot be empty.'; return; }
    if (isNaN(rails) || rails < 2)    { errEl.textContent = '[!] Rails must be >= 2.'; return; }
    if (rails >= text.length)         { errEl.textContent = `[!] Rails must be less than text length (${text.length}).`; return; }
    const cipher = encrypt(text, rails);
    document.getElementById('enc-output').textContent   = cipher;
    document.getElementById('enc-viz-grid').innerHTML   = buildViz(text, rails);
    document.getElementById('enc-rail-legend').innerHTML = buildLegend(rails);
    document.getElementById('enc-result').style.display = '';
  }

  // ── Run decrypt ─────────────────────────────────
  function runDecrypt() {
    const text  = document.getElementById('dec-input').value.trim();
    const rails = parseInt(document.getElementById('dec-rails').value);
    const errEl = document.getElementById('dec-error');
    errEl.textContent = '';
    if (!text)                        { errEl.textContent = '[!] Ciphertext cannot be empty.'; return; }
    if (isNaN(rails) || rails < 2)    { errEl.textContent = '[!] Rails must be >= 2.'; return; }
    if (rails >= text.length)         { errEl.textContent = `[!] Rails must be less than text length (${text.length}).`; return; }
    const plain = decrypt(text, rails);
    document.getElementById('dec-output').textContent   = plain;
    document.getElementById('dec-result').style.display = '';
  }

  // ── Copy to clipboard ───────────────────────────
  function copyText(id, btn) {
    const txt = document.getElementById(id).textContent;
    navigator.clipboard.writeText(txt).then(() => {
      const orig = btn.textContent;
      btn.textContent = 'Copied!';
      setTimeout(() => { btn.textContent = orig; }, 1500);
    }).catch(() => {});
  }
</script>
